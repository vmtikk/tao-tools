--- backfill powershell scripti

Get-Content .env | ForEach-Object {
    if ($_ -match '^\s*#' -or $_ -match '^\s*$') { return }
    $name, $value = $_ -split '=', 2
    Set-Item -Path "env:$($name.Trim())" -Value $value.Trim()
}

$env:FROM_BLOCK = "1"
$env:CHAIN_CHUNK_BLOCKS = "20000"
$env:CHAIN_MAX_RPM = "11500"
$env:CHAIN_CONCURRENCY = "50"
pnpm.cmd --filter @tao-tools/ingest run chain:backfill *>&1 | Tee-Object -FilePath backfill.log
